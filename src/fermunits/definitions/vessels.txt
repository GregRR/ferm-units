# Brewing vessel volumes

# Pint already defines:
# beer_barrel = 31 US liquid gallons
# imperial_gallon = 4.54609 liters
# hogshead = 63 US liquid gallons, traditionally a wine hogshead

# Explicit brewing-qualified alias for Pint's 36-Imperial-gallon barrel
imperial_beer_barrel = imperial_barrel
kilderkin = 18 * imperial_gallon
firkin = 9 * imperial_gallon
pin_cask = 4.5 * imperial_gallon

# Pint's bare hogshead remains available as 63 US liquid gallons. FermUnits
# does not add a generic wine-qualified alias because legitimate wine meanings
# vary by region.
brewing_hogshead = 54 * imperial_gallon

brewing_puncheon = 72 * imperial_gallon
brewing_butt = 108 * imperial_gallon
brewing_tun = 216 * imperial_gallon

# Explicit alias for Pint's 31-US-gallon beer barrel
us_beer_barrel = beer_barrel
